```ts
/**
 * Formatting for the portfolio feature. "en-US" is pinned (not the device
 * locale) so digits stay Latin and grouping stays "1,234" — the same text the
 * web renders and what the tests can assert.
 */

const MONTHS = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];

export function egp(value: number, digits = 0): string {
  return value.toLocaleString("en-US", { maximumFractionDigits: digits });
}

export function egp2(value: number): string {
  return value.toLocaleString("en-US", { minimumFractionDigits: 2, maximumFractionDigits: 2 });
}

export function signed(value: number, digits = 0): string {
  return `${value >= 0 ? "+" : ""}${egp(value, digits)}`;
}

export function signedPct(value: number, digits = 1): string {
  return `${value >= 0 ? "+" : ""}${value.toFixed(digits)}%`;
}

/** P&L is a real direction, so it may use gain/loss. Nothing else should call this. */
export function pnlTone(value: number): "text-gain" | "text-loss" {
  return value >= 0 ? "text-gain" : "text-loss";
}

function pad(n: number): string {
  return n < 10 ? `0${n}` : String(n);
}

/** Local calendar day. Never toISOString(): that is the UTC day. */
export function dateToIso(d: Date): string {
  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`;
}

export function todayIso(now: Date = new Date()): string {
  return dateToIso(now);
}

/** "2026-06-01" -> local midnight of that day; null for anything else. */
export function isoToDate(iso: string): Date | null {
  const m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(iso);
  if (!m) return null;
  const d = new Date(Number(m[1]), Number(m[2]) - 1, Number(m[3]));
  return Number.isNaN(d.getTime()) ? null : d;
}

/** "2026-04-07" -> "7 Apr 2026", for the last-market-dividend line. */
export function fmtExDate(iso: string): string {
  const d = isoToDate(iso);
  if (!d) return iso;
  return `${d.getDate()} ${MONTHS[d.getMonth()]} ${d.getFullYear()}`;
}

/** The shared API throws Error(detail) — show that text, it names the problem. */
export function errorMessage(e: unknown, fallback = "Something went wrong"): string {
  return e instanceof Error && e.message ? e.message : fallback;
}
```

```ts
/**
 * Parsing for the three money fields (buy price, sell price, dividend amount)
 * and the two share fields (quantity, dividend shares).
 *
 * EGX quotes to THREE decimals (CIEB at 27.697), and the dividend amount is
 * whatever landed after 5-10% withholding, so it is rarely round. The web
 * shipped `step={0.01}` once and the browser rejected real prices; this is the
 * mobile equivalent of `step="any"` plus an explicit 3-decimal ceiling.
 *
 * EGX has no fractional shares, so share fields are integers — never
 * parseInt("5.7"), which would silently record 5.
 */

export const MONEY_DECIMALS = 3;

/** `error: null` means "empty — say nothing yet". */
export type Parsed<T> = { ok: true; value: T } | { ok: false; error: string | null };

const ARABIC_INDIC_ZERO = 0x0660;
const EXTENDED_ARABIC_INDIC_ZERO = 0x06f0;
const ARABIC_DECIMAL_SEPARATOR = "٫";
const ARABIC_THOUSANDS_SEPARATOR = "٬";

/** Arabic-Indic and Persian digits -> Latin; Arabic decimal separator -> ".". */
export function normalizeNumeric(text: string): string {
  let out = "";
  for (const ch of text.trim()) {
    const code = ch.charCodeAt(0);
    if (code >= ARABIC_INDIC_ZERO && code <= ARABIC_INDIC_ZERO + 9) {
      out += String(code - ARABIC_INDIC_ZERO);
    } else if (code >= EXTENDED_ARABIC_INDIC_ZERO && code <= EXTENDED_ARABIC_INDIC_ZERO + 9) {
      out += String(code - EXTENDED_ARABIC_INDIC_ZERO);
    } else if (ch === ARABIC_DECIMAL_SEPARATOR) {
      out += ".";
    } else {
      out += ch;
    }
  }
  return out;
}

export function parseMoney(text: string): Parsed<number> {
  const s = normalizeNumeric(text);
  if (s === "") return { ok: false, error: null };
  // "1,200" is 1200 to one reader and 1.2 to another. Guessing wrong records a
  // price off by a factor of a thousand, so a comma is refused, not read.
  if (s.includes(",") || s.includes(ARABIC_THOUSANDS_SEPARATOR)) {
    return { ok: false, error: "Use a dot for decimals and no thousands separators" };
  }
  if (!/^(\d+\.?\d*|\.\d+)$/.test(s)) return { ok: false, error: "Enter a number, e.g. 27.697" };
  const decimals = (s.split(".")[1] ?? "").replace(/0+$/, "").length;
  if (decimals > MONEY_DECIMALS) return { ok: false, error: "EGX prices go to 3 decimals at most" };
  const value = Number(s);
  if (!(value > 0)) return { ok: false, error: "Must be more than 0" };
  return { ok: true, value };
}

export function parseShares(text: string, max?: number): Parsed<number> {
  const s = normalizeNumeric(text);
  if (s === "") return { ok: false, error: null };
  if (!/^\d+$/.test(s)) return { ok: false, error: "Enter a whole number of shares" };
  const value = Number(s);
  if (max !== undefined && (value < 1 || value > max)) {
    return { ok: false, error: `Enter between 1 and ${max} shares` };
  }
  if (value < 1) return { ok: false, error: "Must be at least 1 share" };
  return { ok: true, value };
}
```

```ts
/**
 * Add/Edit holding form rules. The form is deliberately down to symbol,
 * quantity, buy price and buy date (web, 2026-09-01). Target, stop-loss and
 * notes are still stored and still drive signals, so an edit carries them
 * through untouched — nothing here can set or clear them.
 */
import { TICKER_SEARCH_LIMIT } from "@/shared/constants";
import type { PortfolioHolding, Ticker } from "@/shared/types";
import { todayIso } from "./format";
import { parseMoney, parseShares } from "./money";

export type TickerValidation = "idle" | "validating" | "valid" | "invalid" | "error";

export interface ResolvedTicker {
  symbol: string;
  name: string;
  sector: string;
}

export interface HoldingInitialValues {
  symbol: string;
  name?: string;
  sector?: string;
  buy_price?: number;
  buy_date?: string;
  quantity?: number;
  target_price?: number | null;
  stop_loss?: number | null;
  notes?: string;
}

export type HoldingPayload = Omit<PortfolioHolding, "id" | "created_at" | "updated_at">;

export function filterTickers(tickers: Ticker[], search: string, limit = TICKER_SEARCH_LIMIT): Ticker[] {
  const q = search.trim().toLowerCase();
  if (!q) return tickers.slice(0, limit);
  return tickers
    .filter((t) => t.symbol.toLowerCase().includes(q) || t.name.toLowerCase().includes(q))
    .slice(0, limit);
}

export function findKnownTicker(tickers: Ticker[], text: string): Ticker | undefined {
  const sym = text.trim().toUpperCase();
  if (!sym) return undefined;
  return tickers.find((t) => t.symbol.toUpperCase() === sym);
}

/**
 * The ticker an edit starts from. When the ticker list has not loaded (or does
 * not carry the symbol) it falls back to what the HOLDING stores. The web form
 * fell back to name = symbol, sector = "Unknown" here, and the edit then wrote
 * those over the stored company name and sector.
 */
export function initialTicker(tickers: Ticker[], initial?: HoldingInitialValues): ResolvedTicker | null {
  if (!initial?.symbol) return null;
  const known = findKnownTicker(tickers, initial.symbol);
  if (known) return { symbol: known.symbol, name: known.name, sector: known.sector };
  return {
    symbol: initial.symbol,
    name: initial.name || initial.symbol,
    sector: initial.sector || "Unknown",
  };
}

export interface HoldingFormErrors {
  quantity: string | null;
  price: string | null;
  date: string | null;
}

export function buildHoldingPayload(input: {
  ticker: ResolvedTicker | null;
  quantityText: string;
  priceText: string;
  buyDate: string;
  initial?: HoldingInitialValues;
  today?: string;
}): { ok: true; payload: HoldingPayload } | { ok: false; errors: HoldingFormErrors } {
  const quantity = parseShares(input.quantityText);
  const price = parseMoney(input.priceText);
  const today = input.today ?? todayIso();
  const date = !input.buyDate
    ? "Pick a buy date"
    : input.buyDate > today
      ? "The buy date cannot be in the future"
      : null;

  if (!input.ticker || !quantity.ok || !price.ok || date) {
    return {
      ok: false,
      errors: {
        quantity: quantity.ok ? null : quantity.error,
        price: price.ok ? null : price.error,
        date,
      },
    };
  }

  return {
    ok: true,
    payload: {
      symbol: input.ticker.symbol,
      name: input.ticker.name,
      sector: input.ticker.sector,
      buy_price: price.value,
      buy_date: input.buyDate,
      quantity: quantity.value,
      target_price: input.initial?.target_price ?? null,
      stop_loss: input.initial?.stop_loss ?? null,
      notes: input.initial?.notes ?? "",
    },
  };
}
```

```ts
/**
 * Dividend form rules. A dividend is anchored to the SYMBOL (no holding_id),
 * so it can be recorded against a position already sold out — the options are
 * open holdings plus every symbol with a sale. `amount` is what actually
 * landed, net of withholding tax; `shares` is optional and display-only.
 */
import type { PortfolioHolding, Sale } from "@/shared/types";
import { todayIso } from "./format";
import { parseMoney, parseShares } from "./money";

export interface DividendSymbolOption {
  symbol: string;
  name: string;
  sector: string;
  /** Shares held now, for an open position; null for a sold-out symbol. */
  shares: number | null;
}

export function dividendSymbolOptions(
  sales: Sale[] = [],
  holdings: PortfolioHolding[] = [],
): DividendSymbolOption[] {
  const map = new Map<string, DividendSymbolOption>();
  for (const s of sales) {
    map.set(s.symbol, { symbol: s.symbol, name: s.name, sector: s.sector, shares: null });
  }
  // Open holdings win, with the POSITION's share count — the sum of its lots.
  // (The web set it from whichever lot came last, i.e. one purchase's count.)
  const open = new Map<string, DividendSymbolOption>();
  for (const h of holdings) {
    const prev = open.get(h.symbol);
    open.set(h.symbol, {
      symbol: h.symbol,
      name: h.name || h.symbol,
      sector: h.sector ?? "",
      shares: (prev?.shares ?? 0) + h.quantity,
    });
  }
  for (const [symbol, option] of open) map.set(symbol, option);
  return [...map.values()].sort((a, b) => a.symbol.localeCompare(b.symbol));
}

export function presetSharesText(options: DividendSymbolOption[], symbol: string): string {
  const shares = options.find((o) => o.symbol === symbol)?.shares;
  return shares != null ? String(shares) : "";
}

export interface DividendPayload {
  symbol: string;
  name: string;
  sector: string;
  amount: number;
  pay_date: string;
  shares: number | null;
  notes: string;
}

export interface DividendFormErrors {
  symbol: string | null;
  amount: string | null;
  shares: string | null;
  date: string | null;
}

export function buildDividendPayload(input: {
  symbol: string;
  options: DividendSymbolOption[];
  amountText: string;
  payDate: string;
  sharesText: string;
  notes: string;
  today?: string;
}): { ok: true; payload: DividendPayload } | { ok: false; errors: DividendFormErrors } {
  const amount = parseMoney(input.amountText);
  const sharesBlank = input.sharesText.trim() === "";
  const shares = sharesBlank ? null : parseShares(input.sharesText);
  const today = input.today ?? todayIso();
  const errors: DividendFormErrors = {
    symbol: input.symbol ? null : "Pick a stock",
    amount: amount.ok ? null : amount.error,
    shares: shares && !shares.ok ? shares.error : null,
    date: !input.payDate
      ? "Pick a pay date"
      : input.payDate > today
        ? "The pay date cannot be in the future"
        : null,
  };
  if (!input.symbol || !amount.ok || (shares && !shares.ok) || errors.date) {
    return { ok: false, errors };
  }
  const picked = input.options.find((o) => o.symbol === input.symbol);
  return {
    ok: true,
    payload: {
      symbol: input.symbol,
      name: picked?.name ?? input.symbol,
      sector: picked?.sector ?? "",
      amount: amount.value,
      pay_date: input.payDate,
      shares: shares && shares.ok ? shares.value : null,
      notes: input.notes,
    },
  };
}

export function perShareAmount(amountText: string, sharesText: string): number | null {
  const amount = parseMoney(amountText);
  const shares = parseShares(sharesText);
  return amount.ok && shares.ok ? amount.value / shares.value : null;
}
```

```ts
/**
 * A sale comes out of the POSITION and may span several purchase lots. The
 * backend consumes them oldest-first and re-plans authoritatively; this is the
 * preview, and the realized figure MUST be summed over the parts:
 * (price - average) x qty is only right when the sale consumes every lot.
 */
import { planSale, type SalePart } from "@/shared/positions";
import type { PortfolioHolding } from "@/shared/types";
import { parseMoney, parseShares, type Parsed } from "./money";

export interface SellPosition {
  symbol: string;
  name: string;
  /** Every open lot of the symbol, oldest purchase first. */
  lots: SalePart[];
}

export interface SaleSubmit {
  quantity: number;
  sell_price: number;
  sell_date: string;
  notes: string;
}

export function sellPositionFor(
  holdings: PortfolioHolding[] | undefined,
  symbol: string | null,
): SellPosition | null {
  if (!symbol) return null;
  const rows = (holdings ?? []).filter((h) => h.symbol === symbol && h.quantity > 0 && h.id);
  if (!rows.length) return null;
  const lots = rows
    .map((h) => ({ id: h.id, quantity: h.quantity, buy_price: h.buy_price, buy_date: h.buy_date }))
    .sort((a, b) => (a.buy_date || "").localeCompare(b.buy_date || ""));
  return { symbol, name: rows[0].name || symbol, lots };
}

export interface SalePartPreview extends SalePart {
  /** Realized on this part at the typed price; null until a price is valid. */
  pnl: number | null;
}

export interface SalePreview {
  held: number;
  avgBuyPrice: number;
  multiLot: boolean;
  quantity: Parsed<number>;
  price: Parsed<number>;
  parts: SalePartPreview[];
  pnl: number | null;
  pnlPct: number | null;
  /** The newest purchase this sale reaches; the backend rejects anything earlier. */
  earliestSellDate: string;
}

export function previewSale(position: SellPosition, quantityText: string, priceText: string): SalePreview {
  const held = position.lots.reduce((n, l) => n + l.quantity, 0);
  const cost = position.lots.reduce((n, l) => n + l.buy_price * l.quantity, 0);
  const quantity = parseShares(quantityText, held);
  const price = parseMoney(priceText);

  const plan = quantity.ok ? planSale(position.lots, quantity.value) : [];
  const parts: SalePartPreview[] = plan.map((part) => ({
    ...part,
    pnl: price.ok ? (price.value - part.buy_price) * part.quantity : null,
  }));
  const pnl = quantity.ok && price.ok ? parts.reduce((n, p) => n + (p.pnl ?? 0), 0) : null;
  const planCost = plan.reduce((n, p) => n + p.buy_price * p.quantity, 0);

  const earliestSellDate = plan.length
    ? plan.reduce((d, p) => (p.buy_date > d ? p.buy_date : d), "")
    : position.lots.reduce((d, l) => (!d || l.buy_date < d ? l.buy_date : d), "");

  return {
    held,
    avgBuyPrice: held > 0 ? cost / held : 0,
    multiLot: position.lots.length > 1,
    quantity,
    price,
    parts,
    pnl,
    pnlPct: pnl !== null && planCost > 0 ? (pnl / planCost) * 100 : null,
    earliestSellDate,
  };
}

export function sellDateError(
  sellDate: string,
  earliest: string,
  today: string,
  multiLot: boolean,
): string | null {
  if (!sellDate) return "Pick a sell date";
  if (sellDate > today) return "The sell date cannot be in the future";
  if (earliest && sellDate < earliest) {
    return multiLot
      ? `This sale reaches a purchase made on ${earliest} — sell on or after that date`
      : `Sell on or after the buy date (${earliest})`;
  }
  return null;
}
```

```ts
/**
 * Entry/exit zones are graded by confidence. `low` is the leftover bucket —
 * nothing disqualified the zone and nothing confirmed it — so it must not
 * borrow gain-green or loss-red. Both entry AND exit go neutral at low.
 */
import type { EntryExit, ZoneConfidence } from "@/shared/types";

export interface ZonePill {
  kind: "Entry" | "Exit";
  confidence: ZoneConfidence | null;
  direction: "gain" | "loss";
}

export function zonePill(entryExit: EntryExit | null | undefined): ZonePill | null {
  if (!entryExit) return null;
  if (entryExit.entry_zone.active) {
    return { kind: "Entry", confidence: entryExit.entry_zone.confidence, direction: "gain" };
  }
  if (entryExit.exit_zone.active) {
    return { kind: "Exit", confidence: entryExit.exit_zone.confidence, direction: "loss" };
  }
  return null;
}

export function isStrong(confidence: ZoneConfidence | null | undefined): boolean {
  return confidence === "high" || confidence === "medium";
}

export function zoneTone(pill: ZonePill): "neutral" | "gain" | "loss" {
  return isStrong(pill.confidence) ? pill.direction : "neutral";
}

export function zoneLabel(pill: ZonePill): string {
  return `${pill.kind} zone${pill.confidence ? ` · ${pill.confidence}` : ""}`;
}
```

**The FAB computation (exact):**

```
obstruction = max(tabBarHeight, insets.bottom)        // tab bar height already includes insets.bottom;
                                                      // with no bar, the home indicator is the obstruction
overlap     = max(0, sceneBottom − (windowHeight − obstruction))
fabBottom   = overlap + FAB_GAP (12)                  // style.bottom of the absolutely positioned FAB
```
- `tabBarHeight` = `useContext(BottomTabBarHeightContext) ?? 0` (react-navigation's measured bar height, inset included; the context form is used instead of `useBottomTabBarHeight()` because that hook throws outside a navigator, e.g. in tests).
- `insets.bottom` = `useSafeAreaInsets().bottom`; `windowHeight` = `useSafeAreaFrame().height`.
- `sceneBottom` = `y + height` from `measureInWindow` on the screen's root `View`; until measured it is taken as `windowHeight` (the worst case, so the FAB can only start too high, never under the bar).
- In-flow tab bar (scene ends at the bar's top edge): overlap = 0 → bottom = 12. Floating bar (scene runs under it, like the web pill): overlap = tabBarHeight → bottom = tabBarHeight + 12, the mobile spelling of the web's `calc(var(--bottom-nav-clearance) + 12px)`.
- Scroll content ends with a spacer of `fabBottom + FAB_SIZE + 8` so the last card can scroll clear of the FAB.

```ts
export interface DirectionArrow {

/**
 * "Up" means opposite things here: a rising EGX30 is good, a rising USD/EGP
 * is the pound weakening, and the CBE rate is context (neutral).
 */
export function directionArrow(
  direction: string | null,
  opts: { positiveIsGood?: boolean; neutral?: boolean } = {},
): DirectionArrow {
  if (direction !== "up" && direction !== "down") return { glyph: "→", className: "text-white/30" };
  const glyph = direction === "up" ? "↑" : "↓";
  if (opts.neutral) return { glyph, className: "text-white/40" };
  const positiveIsGood = opts.positiveIsGood ?? true;
  const good = direction === "up" ? positiveIsGood : !positiveIsGood;
  return { glyph, className: good ? "text-gain" : "text-loss" };
}

export function macroDetails(data: MacroData): { rate?: string; fx?: string; egx30?: string } {

export const PE_STALE_AFTER_HOURS = 48;

export function peStalenessMessage(status: PEFeedStatus | null, nowMs: number): string | null {
  if (!status?.last_successful_fetch) return null;
  const last = Date.parse(status.last_successful_fetch);
  if (Number.isNaN(last)) return null;
  const ageHours = (nowMs - last) / 36e5;
  if (ageHours < PE_STALE_AFTER_HOURS) return null;
  const days = Math.floor(ageHours / 24);
  const attempt =
    status.last_attempt_status && status.last_attempt_status !== "ok"
      ? ` Last attempt: ${status.last_attempt_status}.`
      : "";
  return `P/E data last updated ${days} day${days === 1 ? "" : "s"} ago — the EGX feed may be temporarily unavailable.${attempt}`;
}

/** Totals cover only priced holdings; silently dropping one used to show a phantom loss. */
export function excludedNotice(m: PortfolioMetrics): string | null {
  const n = m.excluded_count;
  if (!n) return null;
  const basis = m.excluded_invested > 0 ? ` (${egp(m.excluded_invested)} EGP of cost basis)` : "";
  return `${n} holding${n > 1 ? "s" : ""} could not be priced and ${n > 1 ? "are" : "is"} left out of these totals${basis}. The figures below describe the rest of your portfolio.`;
}
```

```ts
/**
 * Page state and FAB placement. See the plan's Task 5 for the derivation; the
 * short version: the FAB clears whatever part of the bottom obstruction (tab
 * bar incl. inset, or the bare home indicator) overlaps this screen.
 */
import type { Portfolio, SalesResponse } from "@/shared/types";

export const FAB_SIZE = 56;
export const FAB_GAP = 12;

export function fabBottomOffset(p: {
  windowHeight: number;
  sceneBottom: number;
  tabBarHeight: number;
  insetBottom: number;
  gap?: number;
}): number {
  const gap = p.gap ?? FAB_GAP;
  const obstruction = Math.max(p.tabBarHeight, p.insetBottom);
  const overlap = Math.max(0, p.sceneBottom - (p.windowHeight - obstruction));
  return Math.round(overlap + gap);
}

export function contentSpacer(fabBottom: number): number {
  return fabBottom + FAB_SIZE + 8;
}

export type PortfolioView = "loading" | "empty" | "sold_out" | "holdings";

export function portfolioView(s: {
  portfolioLoading: boolean;
  portfolio: Portfolio | null;
  salesLoaded: boolean;
  salesError: string | null;
  sales: SalesResponse | null;
}): PortfolioView {
  if (s.portfolioLoading && !s.portfolio) return "loading";
  if ((s.portfolio?.portfolio.length ?? 0) > 0) return "holdings";
  // "Never traded" needs PROOF: sales loaded, no error, and both ledgers empty.
  const neverTraded =
    s.salesLoaded &&
    !s.salesError &&
    s.sales !== null &&
    s.sales.sales.length === 0 &&
    (s.sales.dividends ?? []).length === 0;
  return neverTraded ? "empty" : "sold_out";
}
```

```tsx
/** Best-effort: a failed status fetch shows nothing and never blocks the page. */
export function PEFreshnessBanner() {
  const [status, setStatus] = useState<PEFeedStatus | null>(null);

  useEffect(() => {
    let cancelled = false;
    fetchPEFeedStatus()
      .then((s) => {
        if (!cancelled) setStatus(s);
      })
      .catch(() => {});
    return () => {
      cancelled = true;
    };
  }, []);

  const message = peStalenessMessage(status, Date.now());
  if (!message) return null;
  return (
    <View className="rounded-lg border border-amber-400/30 bg-amber-400/5 px-3 py-2">
      <AppText className="font-sans text-xs text-amber-400">{message}</AppText>
    </View>
  );
}
```

```tsx
import { useState } from "react";
import { Platform, Pressable, View } from "react-native";
import { AppText } from "@/ui";
import { dateToIso, isoToDate } from "../logic/format";

export interface DateFieldProps {
  label: string;
  /** Local ISO date, "YYYY-MM-DD". */
  value: string;
  onChange: (iso: string) => void;
  minimumDate?: string;
  maximumDate?: string;
  hint?: string;
  error?: string | null;
}

export function DateField({ label, value, onChange, minimumDate, maximumDate, hint, error }: DateFieldProps) {
  const [open, setOpen] = useState(false);
  const min = minimumDate ? isoToDate(minimumDate) ?? undefined : undefined;
  const max = maximumDate ? isoToDate(maximumDate) ?? undefined : undefined;

    if (event.type === "set" && date) onChange(dateToIso(date));

  return (
    <View>
      <AppText variant="label" className="mb-1 font-sans text-xs text-white/40">{label}</AppText>

        accessibilityRole="button"
        accessibilityLabel={`${label}: ${value}`}

        onPress={() => setOpen((o) => !o)}
        className={`min-h-[44px] justify-center rounded-lg border bg-white/5 px-3 ${error ? "border-loss/50" : "border-white/10"}`}
      >
        <AppText className="font-mono text-base text-white">{value}</AppText>
      </Pressable>

        <DateTimePicker
          value={isoToDate(value) ?? new Date()}
          mode="date"
          display={Platform.OS === "ios" ? "inline" : "default"}
          minimumDate={min}
          maximumDate={max}
          themeVariant="dark"
```

```tsx
import { useEffect, useMemo, useRef, useState } from "react";
import { ActivityIndicator, Pressable, View } from "react-native";
import { validateTicker } from "@/shared/api";
import { TICKER_VALIDATION_DEBOUNCE_MS } from "@/shared/constants";
import type { Ticker } from "@/shared/types";
import { AppText, Button, TextField } from "@/ui";
import { todayIso } from "../logic/format";
import {
  buildHoldingPayload,
  filterTickers,
  findKnownTicker,
  initialTicker,
  type HoldingInitialValues,
  type HoldingPayload,
  type ResolvedTicker,
  type TickerValidation,
} from "../logic/holdingForm";
import { DateField } from "./DateField";
import { FormError } from "./FormError";
import { FormScroll } from "./FormScroll";

export interface AddHoldingFormProps {
  tickers: Ticker[];
  initialValues?: HoldingInitialValues;
  onSubmit: (payload: HoldingPayload) => Promise<void>;
  onCancel: () => void;
  error?: string | null;
  onDismissError?: () => void;
}

interface RemoteCheck {
  /** The symbol this answer is for — an answer for "ABU" never applies to "ABUK". */
  symbol: string;
  state: "valid" | "invalid" | "error";
  name: string | null;
}

export function AddHoldingForm({ tickers, initialValues, onSubmit, onCancel, error = null, onDismissError }: AddHoldingFormProps) {
  const editing = Boolean(initialValues);
  const [search, setSearch] = useState(initialValues?.symbol ?? "");
  const [picked, setPicked] = useState<ResolvedTicker | null>(() => initialTicker(tickers, initialValues));
  const [remote, setRemote] = useState<RemoteCheck | null>(null);
  const [showList, setShowList] = useState(false);
  const [quantityText, setQuantityText] = useState(initialValues?.quantity?.toString() ?? "");
  const [priceText, setPriceText] = useState(initialValues?.buy_price?.toString() ?? "");
  const [buyDate, setBuyDate] = useState(initialValues?.buy_date ?? todayIso());
  const [submitting, setSubmitting] = useState(false);
  const inFlight = useRef(false);

  const sym = search.trim().toUpperCase();
  const known = picked ? undefined : findKnownTicker(tickers, sym);
  const knownSymbol = known?.symbol ?? null;
  const answer = remote && remote.symbol === sym ? remote : null;

  // Only an unknown, non-empty, un-picked symbol goes to the server.
  useEffect(() => {
    if (picked || !sym || knownSymbol) return;
    let cancelled = false;
    const timer = setTimeout(async () => {
      try {
        const data = await validateTicker(sym);
        if (cancelled) return;
        const state = data.valid === true ? "valid" : data.valid === false ? "invalid" : "error";
        setRemote({ symbol: sym, state, name: data.name });
      } catch {
        // Network inconclusive: let the user proceed; analysis will fail gracefully.
        if (!cancelled) setRemote({ symbol: sym, state: "error", name: null });
      }
    }, TICKER_VALIDATION_DEBOUNCE_MS);
    return () => {
      cancelled = true;
      clearTimeout(timer);
    };
  }, [sym, picked, knownSymbol]);

  const validation: TickerValidation = picked
    ? "valid"
    : !sym
      ? "idle"
      : known
        ? "valid"
        : answer
          ? answer.state
          : "validating";

  const effective: ResolvedTicker | null =
    picked ??
    (known
      ? { symbol: known.symbol, name: known.name, sector: known.sector }
      : answer && answer.state !== "invalid"
        ? { symbol: sym, name: answer.name || sym, sector: "Unknown" }
        : null);

  const suggestions = useMemo(() => filterTickers(tickers, search), [tickers, search]);
  const result = buildHoldingPayload({ ticker: effective, quantityText, priceText, buyDate, initial: initialValues });
  const errors = result.ok ? null : result.errors;
  const canSubmit = result.ok && validation !== "validating" && validation !== "invalid" && !submitting;

  const edited = () => {
    if (error) onDismissError?.();
  };

  const submit = async () => {
    if (!result.ok || !canSubmit || inFlight.current) return;
    inFlight.current = true;
    setSubmitting(true);
    try {
      await onSubmit(result.payload);
    } finally {
      inFlight.current = false;
      setSubmitting(false);
    }
  };

  return (
    <FormScroll>
      <View>
        <TextField
          label="Stock Symbol *"
          value={search}
          onChangeText={(t) => {
            setSearch(t);
            setPicked(null);
            setShowList(true);
            edited();
          }}
          onFocus={() => setShowList(true)}
          placeholder="Search by symbol or name..."
          autoCapitalize="characters"
          autoCorrect={false}
          error={validation === "invalid" ? "Symbol not found on EGX — check and try again" : null}
        />
        {showList && !picked && suggestions.length > 0 && (
          <View className="mt-1 overflow-hidden rounded-lg border border-white/10 bg-charcoal">
            {suggestions.map((t) => (
              <Pressable
                key={t.symbol}
                accessibilityRole="button"
                onPress={() => {
                  setPicked({ symbol: t.symbol, name: t.name, sector: t.sector });
                  setSearch(t.symbol);
                  setShowList(false);
                  edited();
                }}
                className="min-h-[44px] flex-row items-center gap-3 px-3"
              >
                <AppText className="font-mono-semibold text-xs text-white">{t.symbol}</AppText>
                <AppText numberOfLines={1} className="flex-1 font-sans text-xs text-white/40">{t.name}</AppText>
                <AppText className="font-sans text-[10px] text-accent">{t.sector}</AppText>
              </Pressable>
            ))}
          </View>
        )}
        {validation === "validating" && (
          <View className="mt-1 flex-row items-center gap-2">
            <ActivityIndicator size="small" />
            <AppText className="font-sans text-[10px] text-white/40">Checking…</AppText>
          </View>
        )}
        {validation === "valid" && effective && (
          <AppText className="mt-1 font-sans text-[10px] text-white/60">
            {`${effective.name}${effective.sector !== "Unknown" ? ` — ${effective.sector}` : ""}`}
          </AppText>
        )}
        {validation === "error" && (
          <AppText className="mt-1 font-sans text-[10px] text-white/30">Could not verify — will attempt to load data anyway</AppText>
        )}
      </View>

      <TextField
        label="Quantity *"
        value={quantityText}
        onChangeText={(t) => {
          setQuantityText(t);
          edited();
        }}
        keyboardType="number-pad"
        placeholder="100"
        error={errors?.quantity ?? null}
      />
      <TextField
        label="Buy Price (EGP) *"
        value={priceText}
        onChangeText={(t) => {
          setPriceText(t);
          edited();
        }}
        keyboardType="decimal-pad"
        placeholder="85.50"
        error={errors?.price ?? null}
      />
      <DateField
        label="Buy Date"
        value={buyDate}
        maximumDate={todayIso()}
        onChange={(d) => {
          setBuyDate(d);
          edited();
        }}
        error={errors?.date ?? null}
      />

      <FormError message={error} />

      <View className="flex-row gap-3">
        <View className="flex-1">
          <Button
            label={editing ? "Update" : "Add to Portfolio"}
            onPress={submit}
            disabled={!canSubmit}
            loading={submitting}
          />
        </View>
        <View className="flex-1">
          <Button label="Cancel" variant="secondary" onPress={onCancel} />
        </View>
      </View>
    </FormScroll>
  );
}
```
Note: with `validation === "invalid"` the not-found message is the symbol field's `error`, so the test's `findByText("Symbol not found on EGX — check and try again")` finds the `TextField` error text.

```ts
  salesLoaded: boolean;
  salesError: string | null;
  analyze(): Promise<void>;
  loadSales(): Promise<void>;
  refresh(): Promise<void>;
  addHolding(payload: HoldingPayload): Promise<void>;
  updateHolding(id: string, payload: HoldingPayload): Promise<void>;
  recordSale(position: SellPosition, data: SaleSubmit): Promise<void>;
  recordDividend(payload: DividendPayload): Promise<void>;
  deleteHolding(id: string): Promise<void>;
  deleteSale(id: string): Promise<void>;
  deleteDividend(id: string): Promise<void>;
}

export function usePortfolioData(weightsVersion: number): PortfolioData {
  const [portfolio, setPortfolio] = useState<Portfolio | null>(null);
  const [portfolioLoading, setPortfolioLoading] = useState(true);
  const [analysis, setAnalysis] = useState<PortfolioAnalysisResponse | null>(null);
  const [analyzing, setAnalyzing] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [sales, setSales] = useState<SalesResponse | null>(null);
  const [salesLoaded, setSalesLoaded] = useState(false);
  const [salesError, setSalesError] = useState<string | null>(null);

  const loadPortfolio = useCallback(async () => {
    setPortfolioLoading(true);
    try {
      setPortfolio(await fetchPortfolio());
    } catch (e) {
      setError(errorMessage(e, "Could not load your portfolio"));
    } finally {
      setPortfolioLoading(false);
    }
  }, []);

  // Separate from analysis: realized gains need no price fetch, and a sales
  // outage must not take the page down — it gets its own retry line.
  const loadSales = useCallback(async () => {
    try {
      setSales(await fetchSales());
      setSalesError(null);
    } catch (e) {
      setSalesError(errorMessage(e, "Could not load closed positions"));
    } finally {
      setSalesLoaded(true);
    }
  }, []);

  const analyze = useCallback(async () => {
    setAnalyzing(true);
    setError(null);
    try {
      setAnalysis(await fetchPortfolioAnalysis());
    } catch (e) {
      setError(errorMessage(e, "Could not analyse your portfolio"));
    } finally {
      setAnalyzing(false);
    }
  }, []);

  useEffect(() => {
    void loadPortfolio();
    void loadSales();
  }, [loadPortfolio, loadSales]);

  // Analyse whenever the analysis has been cleared (first load, or after a
  // mutation) and there is something to analyse.
  useEffect(() => {
    if (portfolio && portfolio.portfolio.length > 0 && analysis === null) void analyze();
  }, [portfolio, analysis, analyze]);

  // A mutation raises `analyzing` before the portfolio reloads. If it reloads
  // EMPTY (sold out) no analysis will run to lower it, so the flag is only
  // reported while there is something to analyse — otherwise a stuck spinner
  // and a dead refresh button are all the sold-out page would show. Derived
  // rather than reset inside an effect.
  const hasHoldings = (portfolio?.portfolio.length ?? 0) > 0;

  // New score weights => new scores. Skips the mount (handled above).
  const seenVersion = useRef(weightsVersion);
  useEffect(() => {
    if (seenVersion.current === weightsVersion) return;
    seenVersion.current = weightsVersion;
    if (portfolio && portfolio.portfolio.length > 0) void analyze();
  }, [weightsVersion, portfolio, analyze]);

  const refreshAfterMutation = useCallback(async () => {
    setAnalysis(null);
    setAnalyzing(true);
    await loadPortfolio();
  }, [loadPortfolio]);

  const refresh = useCallback(async () => {
    if (portfolio && portfolio.portfolio.length > 0) {
      await Promise.all([refreshAfterMutation(), loadSales()]);
    } else {
      await Promise.all([loadPortfolio(), loadSales()]);
    }
  }, [portfolio, refreshAfterMutation, loadPortfolio, loadSales]);

  const addHolding = useCallback(
    async (payload: HoldingPayload) => {
      await apiAddHolding(payload);
      await refreshAfterMutation();
    },
    [refreshAfterMutation],
  );

  const updateHolding = useCallback(
    async (id: string, payload: HoldingPayload) => {
      await apiUpdateHolding(id, payload);
      await refreshAfterMutation();
    },
    [refreshAfterMutation],
  );

  const recordSale = useCallback(
    async (position: SellPosition, data: SaleSubmit) => {
      await apiRecordSale({ holding_id: position.lots[0].id, ...data });
      await Promise.all([refreshAfterMutation(), loadSales()]);
    },
    [refreshAfterMutation, loadSales],
  );

  const recordDividend = useCallback(
    async (payload: DividendPayload) => {
      await apiRecordDividend(payload);
      await Promise.all([refreshAfterMutation(), loadSales()]);
    },
    [refreshAfterMutation, loadSales],
  );

  const deleteHolding = useCallback(
    async (id: string) => {
      try {
        await apiDeleteHolding(id);
        await refreshAfterMutation();
      } catch (e) {
        setError(errorMessage(e));
      }
    },
    [refreshAfterMutation],
  );

  const deleteSale = useCallback(
    async (id: string) => {
      try {
        await apiDeleteSale(id);
        await Promise.all([refreshAfterMutation(), loadSales()]);
      } catch (e) {
        setError(errorMessage(e));
      }
    },
    [refreshAfterMutation, loadSales],
  );

  const deleteDividend = useCallback(
    async (id: string) => {
      try {
        await apiDeleteDividend(id);
        await Promise.all([refreshAfterMutation(), loadSales()]);
      } catch (e) {
        setError(errorMessage(e));
      }
    },
    [refreshAfterMutation, loadSales],
  );

  return {
    portfolio, portfolioLoading, analysis, analyzing: analyzing && hasHoldings, error, sales, salesLoaded, salesError,
    analyze, loadSales, refresh, addHolding, updateHolding, recordSale, recordDividend,
    deleteHolding, deleteSale, deleteDividend,
  };
}
```

```tsx
import { BottomTabBarHeightContext } from "@react-navigation/bottom-tabs";
import { useContext } from "react";
import { Pressable } from "react-native";
import { useSafeAreaFrame, useSafeAreaInsets } from "react-native-safe-area-context";
import { AppText } from "@/ui";
import { FAB_SIZE, fabBottomOffset } from "./logic/layout";

/**
 * style.bottom for the FAB. The context form of the tab-bar height is used
 * (not useBottomTabBarHeight) because the hook throws outside a navigator.
 * Until the scene has been measured, assume it reaches the window bottom —
 * the worst case, so the FAB can only start too high, never under the bar.
 */
export function useFabBottom(sceneBottom: number | null): number {
  const tabBarHeight = useContext(BottomTabBarHeightContext) ?? 0;
  const insets = useSafeAreaInsets();
  const frame = useSafeAreaFrame();
  return fabBottomOffset({
    windowHeight: frame.height,
    sceneBottom: sceneBottom ?? frame.height,
    tabBarHeight,
    insetBottom: insets.bottom,
  });
}

export function AddFab({ bottom, onPress }: { bottom: number; onPress: () => void }) {
  return (
    <Pressable
      accessibilityRole="button"
      accessibilityLabel="Add a stock"
      onPress={onPress}
      style={{ position: "absolute", right: 16, bottom, width: FAB_SIZE, height: FAB_SIZE }}
      className="items-center justify-center rounded-full bg-accent shadow-lg active:opacity-80"
    >
      <AppText className="font-sans-bold text-2xl text-charcoal-dark">+</AppText>
    </Pressable>
  );
}
```
